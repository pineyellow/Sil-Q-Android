cmake_minimum_required(VERSION 3.22.1)
project(silq_android C)
option(SILQ_DEBUG_TUTORIAL_SKILLS "Boost tutorial skills and XP in Debug builds" OFF)
get_filename_component(SIL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${SIL_ROOT}/android/vendor/SDL2" SDL2-build)

# Include the shared engine source lists directly.
include("${SIL_ROOT}/src/core-sources.cmake")
set(CORE_SOURCES "")
foreach(SOURCE IN LISTS Z_SOURCES SIL_SOURCES)
    list(APPEND CORE_SOURCES "${SIL_ROOT}/${SOURCE}")
endforeach()
add_library(silq SHARED ${CORE_SOURCES} "${SIL_ROOT}/src/main.c" main-sdl.c save-sdl.c)
target_include_directories(silq PRIVATE "${SIL_ROOT}/src")
target_compile_features(silq PRIVATE c_std_99)
target_compile_options(silq PRIVATE -Wall -Wextra -fsigned-char)
target_compile_definitions(silq PRIVATE USE_SDL HAVE_MKSTEMP)
# Even an explicitly enabled option cannot add the boost to a release build.
if(SILQ_DEBUG_TUTORIAL_SKILLS)
    target_compile_definitions(silq PRIVATE $<$<CONFIG:Debug>:SILQ_DEBUG_TUTORIAL_SKILLS>)
endif()
set_source_files_properties("${SIL_ROOT}/src/main.c" PROPERTIES COMPILE_DEFINITIONS "main=sil_main")
target_link_libraries(silq PRIVATE SDL2 android log m)
